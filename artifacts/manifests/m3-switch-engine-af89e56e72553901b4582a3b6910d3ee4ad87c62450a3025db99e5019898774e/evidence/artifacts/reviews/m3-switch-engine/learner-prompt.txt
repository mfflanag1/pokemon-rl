30: # ----------------------------------------------------------------------------- masked PPO loss
31: 
32: def masked_loss(logits, values, mask, actions, old_logp, advantages, returns, valid,
33:                 clip=.2, value_coef=.5, entropy_coef=.01):
34:     """Padded PPO loss -> (scalar_loss, metrics) over the rows flagged by `valid`.
35: 
36:     Shapes (n rows, A actions): `logits` (n, A) FP32, `values` (n,) FP32, `mask` (n, A) bool,
37:     `actions` (n,) integer, `old_logp`/`advantages`/`returns` (n,) FP32, `valid` (n,) bool.
38:     `clip`, `value_coef`, `entropy_coef` are Python floats (host-rounded to FP32 like the
39:     reference's `F32(1 - clip)` / `F32(1 + clip)`); treat them as static under `jax.jit`.
40: 
41:     Semantics on the valid rows are exactly those of the sealed
42:     `pokemon_jax_learner.core._loss` applied to the compacted valid subset:
43:     mean clipped actor loss with the explicit inclusive `active` branch (equality on the
44:     unclipped side keeps the gradient), `value_coef` * half-MSE value loss, minus
45:     `entropy_coef` * mean entropy; `old_logp`, `advantages`, `returns` are stop-gradient.
46:     Metrics are the reference's seven keys: `loss`, `actor_loss`, `value_loss`, `entropy`,
47:     `max_ratio_error` (max over valid rows), `clip_fraction`, `approx_kl` (valid-row means).
48: 
49:     Invalid rows may hold any transport payload (NaN logits/values/targets, action -1,
50:     all-false mask). Every input is sanitized with `jnp.where(valid, ...)` to neutral finite
51:     values (zeros; all-true mask; action 0) *before* any arithmetic, and every per-row term is
52:     re-masked before its reduction, so invalid rows contribute neither value nor gradient:
53:     both selections have exact-zero cotangents, giving exactly 0 gradient entries on padded
54:     rows. With no valid rows the divisor is 1, so all seven metrics and both gradients are
55:     exactly zero. Differentiable in `logits` and `values`; no host checks.
56:     """
57:     valid = jnp.asarray(valid, bool)
58:     valid_rows = valid[:, None]
59:     zero = jnp.float32(0)
60:     # Sanitize *all* inputs on invalid rows before any arithmetic touches them.
61:     logits = jnp.where(valid_rows, logits, zero)
62:     mask = jnp.where(valid_rows, mask, True)
63:     actions = jnp.where(valid, actions, 0).astype(jnp.int32)
64:     values, old_logp, advantages, returns = [jnp.where(valid, a, zero) for a in (values, old_logp, advantages, returns)]
65:     old_logp, advantages, returns = [jax.lax.stop_gradient(a) for a in (old_logp, advantages, returns)]
66:     low, high, clip, value_coef, entropy_coef = [
67:         jnp.asarray(c, jnp.float32) for c in (1 - clip, 1 + clip, clip, value_coef, entropy_coef)]
68: 
69:     count = jnp.maximum(jnp.sum(valid.astype(jnp.float32)), jnp.float32(1))
70: 
71:     def valid_mean(rows):
72:         return jnp.sum(jnp.where(valid, rows, zero)) / count
73: 
74:     p, logp = distribution_kernel(logits, mask)
75:     n = logits.shape[0]
76:     log_ratio = logp[jnp.arange(n), actions] - old_logp
77:     ratio = jnp.exp(log_ratio)
78:     clipped = jnp.clip(ratio, low, high)
79:     # Reference inclusive branch: equality on the unclipped side keeps the gradient.
80:     active = jnp.where(advantages >= 0, ratio <= high, ratio >= low)
81:     surrogate = jnp.where(active, ratio, jax.lax.stop_gradient(clipped)) * advantages
82:     actor_loss = -valid_mean(surrogate)
83:     entropy = valid_mean(-jnp.sum(p * logp, axis=1))
84:     value_loss = jnp.float32(.5) * valid_mean(jnp.square(values - returns))
85:     loss = actor_loss + value_coef * value_loss - entropy_coef * entropy
86:     error = jnp.abs(ratio - jnp.float32(1))
87:     metrics = {'loss': loss, 'actor_loss': actor_loss, 'value_loss': value_loss, 'entropy': entropy,
88:                'max_ratio_error': jnp.max(jnp.where(valid, error, zero)),
89:                'clip_fraction': valid_mean((error > clip).astype(jnp.float32)),
90:                'approx_kl': valid_mean((ratio - jnp.float32(1)) - log_ratio)}
91:     return loss, metrics

94: # ----------------------------------------------------------------------------- bias correction
95: 
96: BIAS_TABLE_STEPS = 20000
97: # Host-rounded exactly like the sealed `Adam.apply`: F32(1 - beta ** step) from Python-double
98: # `beta ** step`. Row s-1 holds step s. Every step beyond the table has 0.999 ** step < 2**-25,
99: # so its FP32 rounding is exactly 1.0 (and 0.9 ** step underflowed long before).
100: _BIAS_TABLE = np.array([[1 - .9 ** s, 1 - .999 ** s] for s in range(1, BIAS_TABLE_STEPS + 1)], np.float32)
101: 
102: 
103: def bias_correction(step):
104:     """FP32 pair `[1 - 0.9**step, 1 - 0.999**step]` for a positive int32 `step`, shape (2,).
105: 
106:     Steps 1..`BIAS_TABLE_STEPS` are read from the precomputed NumPy host-rounded table (the
107:     sealed reference constants bit for bit); later steps return exact FP32 ones, which is the
108:     host rounding for every such step. Non-positive steps clamp to the step-1 entry (the
109:     contract only covers positive steps). Pure gather/select: safe under `jit` and `scan`.
110:     """
111:     step = jnp.asarray(step, jnp.int32)
112:     index = jnp.clip(step - 1, 0, BIAS_TABLE_STEPS - 1)
113:     tabled = jnp.asarray(_BIAS_TABLE)[index]
114:     return jnp.where(step > BIAS_TABLE_STEPS, jnp.ones(2, jnp.float32), tabled)
115:

119: def adam_step(parameters, gradients, m, v, step, lr=.003, max_norm=1.):
120:     """Functional sealed Adam step -> (new_parameters, new_m, new_v, new_step, preclip_norm).
121: 
122:     `parameters`, `gradients`, `m`, `v` are same-keyed dicts of FP32 arrays (`m`/`v` zeros
123:     before the first update); `step` is the int32 number of updates already applied (0 at
124:     start) and is returned incremented by one. The recipe is the sealed
125:     `pokemon_jax_learner.core.Adam.apply`: global-norm clip to `max_norm` (FP32 norm
126:     returned pre-clip), beta 0.9/0.999, eps 1e-8, bias correction at `step + 1` from
127:     `bias_correction`, via the sealed `_adam_kernel`. Python-float `lr`/`max_norm` are
128:     host-rounded to FP32 like the reference. Pure and mutation-free (fresh dicts), so the
129:     tuple can be scanned; no host checks. Callers skip this call for empty minibatches.
130:     """
131:     new_step = jnp.asarray(step, jnp.int32) + 1
132:     bias = bias_correction(new_step)
133:     new_parameters, new_m, new_v, norm = _adam_kernel(
134:         dict(parameters), dict(gradients), dict(m), dict(v),
135:         jnp.asarray(lr, jnp.float32), jnp.asarray(max_norm, jnp.float32), bias[0], bias[1])
136:     return new_parameters, new_m, new_v, new_step, norm
