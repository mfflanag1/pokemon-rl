FILE python/pokemon_binary/policy.py
1: """Two-action policy with exact dyadic word selection, conditional on uniform words.
2: 
3: The rare FP32 mass and its real complement define the law. Logs use log/log1p
4: of that same mass; rounded displayed probabilities never define the sampler.
5: Only this binary profile is admitted. The integer sampler is not differentiated.
6: """
7: import jax
8: import jax.numpy as jnp
9: import numpy as np
10: from pokemon_jax_learner.model import (ActorInput, forward, validate_input,
11:                                        validate_parameters)
12: from pokemon_learner.core import require
13: 
14: FLOOR = np.float32(2**-24)
15: DOMAIN = 0x44594231
16: 
17: 
18: def parts(logits, mask):
19:     both = jnp.all(mask, axis=1)
20:     # Forced/padded rows must have no logit gradient, including their difference.
21:     safe = jnp.where(both[:, None], logits, jnp.float32(0))
22:     delta = safe[:, 0] - safe[:, 1]
23:     rare_action = jnp.where(delta >= 0, 1, 0).astype(jnp.int32)
24:     tail = jnp.exp(-jnp.abs(delta))
25:     raw = tail / (jnp.float32(1) + tail)
26:     rare = jnp.where(raw > FLOOR, raw, FLOOR)  # zero derivative at the floor
27:     is_rare = jnp.arange(2)[None, :] == rare_action[:, None]
28:     p = jnp.where(is_rare, rare[:, None], jnp.float32(1) - rare[:, None])
29:     logs = jnp.where(is_rare, jnp.log(rare)[:, None], jnp.log1p(-rare)[:, None])
30:     # Forced legal action: exact one/zero; terminal: exact all-zero transport.
31:     return (jnp.where(both[:, None], p, mask.astype(jnp.float32)),
32:             jnp.where(both[:, None], logs, jnp.float32(0)), rare_action, rare)
33: 
34: 
35: def distribution_kernel(logits, mask):
36:     return parts(logits, mask)[:2]
37: 
38: 
39: def threshold_words(rare):
40:     """Exact high/low words of rare*2^64 for normal FP32 rare in [2^-24,1/2].
41: 
42:     rare = m * 2^(e-150), m a 24-bit significand and biased exponent e.
43:     The shift e-86 lies in [17,40], so two uint32 words suffice without x64.
44:     """
45:     bits = jax.lax.bitcast_convert_type(rare, jnp.uint32)
46:     mantissa = (bits & jnp.uint32(0x7fffff)) | jnp.uint32(0x800000)
47:     shift = ((bits >> jnp.uint32(23)) & jnp.uint32(255)).astype(jnp.int32) - 86
48:     high = jnp.where(shift < 32,
49:         mantissa >> jnp.maximum(32 - shift, 0).astype(jnp.uint32),
50:         mantissa << jnp.maximum(shift - 32, 0).astype(jnp.uint32))
51:     low = jnp.where(shift < 32, mantissa << jnp.minimum(shift, 31).astype(jnp.uint32), jnp.uint32(0))
52:     return jnp.stack((high, low), axis=-1)
53: 
54: 
55: def select_words(rare, rare_action, mask, words):
56:     """Pure injected-word selector used identically by tests and keyed sampling."""
57:     threshold = threshold_words(rare)
58:     choose_rare = ((words[:, 0] < threshold[:, 0]) |
59:                    ((words[:, 0] == threshold[:, 0]) & (words[:, 1] < threshold[:, 1])))
60:     selected = jnp.where(choose_rare, rare_action, 1 - rare_action)
61:     live = jnp.any(mask, axis=1)
62:     return jnp.where(live, jnp.where(jnp.all(mask, axis=1), selected,
63:                                    jnp.argmax(mask, axis=1)), -1).astype(jnp.int32)
64: 
65: 
66: @jax.jit
67: def predict_kernel(parameters, obs):
68:     logits, values = forward(parameters, obs)
69:     probabilities, logs = distribution_kernel(logits, obs.mask)
70:     return probabilities, logs, jnp.where(jnp.any(obs.mask, axis=1), values, 0)
71: 
72: 
73: @jax.jit
74: def action_kernel(parameters, obs, keys):
75:     logits, values = forward(parameters, obs)
76:     _, logs, rare_action, rare = parts(logits, obs.mask)
77:     words = jax.vmap(lambda key: jax.random.bits(jax.random.fold_in(key, DOMAIN),
78:                                                (2,), dtype=jnp.uint32))(keys)
