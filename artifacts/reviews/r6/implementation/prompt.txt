63: export function pathways(source, root = source.ast) {

66:     let kind, selector, args = [];

70:       kind = Object.hasOwn(SINKS, name) ? SINKS[name] : undefined;
71:       if (ts.isElementAccessExpression(expr)) kind = 'computed-call';
72:       const target = expr.getText(source.ast);
73:       if (name === 'get' && /\.(moves|abilities|items|species|conditions)\.get$/.test(target)) kind = 'dex-resolution';
74:       // Retain every remaining call under the enclosing source obligation.
75:       // Local aliases can themselves hold callbacks, so dropping ordinary
76:       // identifier/property calls would hide potential dynamic dispatch.
77:       if (!kind) kind = 'source-owned-call';
78:       selector = target;
79:       args = n.arguments.map(a => ({expression: a.getText(source.ast), literal: ts.isStringLiteral(a) ? a.text : null}));
80:     } else if (ts.isNewExpression(n)) {
81:       kind = 'source-owned-construction'; selector = n.expression.getText(source.ast);
82:       args = (n.arguments ?? []).map(a => ({expression: a.getText(source.ast), literal: ts.isStringLiteral(a) ? a.text : null}));
83:     } else if (ts.isBinaryExpression(n) && n.operatorToken.kind === ts.SyntaxKind.EqualsToken
84:         && ts.isPropertyAccessExpression(n.left) && STATE_FIELDS.has(keyName(n.left.name))) {
85:       kind = 'identity-state-write'; selector = n.left.getText(source.ast);
86:       args = [{expression: n.right.getText(source.ast), literal: ts.isStringLiteral(n.right) ? n.right.text : null}];

88:     out.push({kind, selector, arguments: args, source: source.ref(n)});
