        and, when ``baseline`` is given, ``improvement={mean, lower, upper}``.
    """
    seed = _validate_seed(seed)
    replicates = _validate_replicates(replicates)
    alpha = _validate_alpha(alpha)

    cand = _validate_scores(candidate, "candidate")
    base: Optional[np.ndarray] = None
    if baseline is not None:
        base = _validate_scores(baseline, "baseline")
        if base.shape != cand.shape:
            raise ValueError(
                f"baseline shape {base.shape} must match candidate shape {cand.shape}"
            )

    # Average the two seats FIRST: the pair is the resampling unit.
    pair_means = cand.mean(axis=1)
    pair_diffs = (pair_means - base.mean(axis=1)) if base is not None else None

    boot_means, boot_diffs = _pair_bootstrap(
        pair_means, pair_diffs, seed=seed, replicates=replicates
    )
    lower, upper = _interval(boot_means, alpha)

    n_pairs = int(cand.shape[0])
    result: Dict[str, Any] = {
        "pairs": n_pairs,
        "games": 2 * n_pairs,
        "wins": int(np.count_nonzero(cand == 1.0)),
        "draws": int(np.count_nonzero(cand == 0.5)),
        "losses": int(np.count_nonzero(cand == 0.0)),
        "mean": float(pair_means.mean()),
        "lower": lower,
        "upper": upper,
    }
    if pair_diffs is not None and boot_diffs is not None:
        d_lower, d_upper = _interval(boot_diffs, alpha)
        result["improvement"] = {
            "mean": float(pair_diffs.mean()),
            "lower": d_lower,
            "upper": d_upper,
        }
    return result


def calibration(values: Any, outcomes: Any) -> Dict[str, Any]:
    """Descriptive calibration of critic value predictions against outcomes.

    Parameters
    ----------
    values
        Non-empty 1-D finite numeric predictions (may fall outside [0, 1]).
    outcomes
        Non-empty 1-D finite numeric array of the same length; entries must be
        exactly ``0``, ``0.5`` or ``1``.

    Returns
    -------
    dict
        ``count``: number of predictions.
        ``raw_mse``: mean squared error of the *unclipped* predictions.
        ``clipped_brier``: mean squared error after clipping predictions to
        ``[0, 1]``.
        ``out_of_range_rate``: fraction of predictions strictly outside
        ``[0, 1]``.
        ``bins``: ten reliability bins; ``bin = floor(clipped * 10)`` capped at
        9. Each bin is ``{index, count, mean_prediction, mean_outcome}`` where
        the means are ``None`` for empty bins. ``mean_prediction`` is the mean
        of the clipped predictions in the bin.
    """
    vals = _as_numeric_array(values, "values")
    outs = _as_numeric_array(outcomes, "outcomes")
    if vals.ndim != 1 or outs.ndim != 1:
        raise ValueError("values and outcomes must be one-dimensional")
    if vals.shape[0] != outs.shape[0]:
        raise ValueError(
            f"values length {vals.shape[0]} must equal outcomes length {outs.shape[0]}"
        )
    if not np.isin(outs, _VALID_SCORES).all():
        raise ValueError("outcomes must be exactly 0, 0.5 or 1")

    count = int(vals.shape[0])
    raw_mse = float(np.mean((vals - outs) ** 2))  # NOT clipped
    clipped = np.clip(vals, 0.0, 1.0)
    clipped_brier = float(np.mean((clipped - outs) ** 2))
    out_of_range = np.count_nonzero((vals < 0.0) | (vals > 1.0))
    out_of_range_rate = float(out_of_range / count)

    bin_index = np.minimum(
        np.floor(clipped * _NUM_BINS).astype(np.int64), _NUM_BINS - 1
    )
    bins: List[Dict[str, Any]] = []
    for b in range(_NUM_BINS):
        mask = bin_index == b
        n = int(np.count_nonzero(mask))
        bins.append(
            {
                "index": b,
                "count": n,
                "mean_prediction": float(clipped[mask].mean()) if n else None,
                "mean_outcome": float(outs[mask].mean()) if n else None,
            }
        )

    return {
        "count": count,
        "raw_mse": raw_mse,
        "clipped_brier": clipped_brier,
        "out_of_range_rate": out_of_range_rate,
        "bins": bins,
    }

import unittest
import numpy as np
from pokemon_pilot.statistics import paired_summary, calibration


class StatisticsTests(unittest.TestCase):
    def test_pair_is_unit_and_seats_are_not_independent(self):
        # Opposite seats cancel exactly within every pair; game resampling would vary.
        x=np.tile([0.,1.],(20,1))
        r=paired_summary(x,seed=7,replicates=512,alpha=.05)
        self.assertEqual((r['mean'],r['lower'],r['upper']),(.5,.5,.5))
        self.assertEqual((r['pairs'],r['games']),(20,40))
        self.assertEqual(r['wins'],20);self.assertEqual(r['losses'],20)

    def test_paired_differences_keep_common_variation(self):
        x=np.array([[0,0],[1,1],[.5,.5]]*10,dtype=float)
        r=paired_summary(x,baseline=x.copy(),seed=3,replicates=512,alpha=.05)
        self.assertLess(r['lower'],r['upper'])
        self.assertEqual(r['improvement'],{'mean':0.,'lower':0.,'upper':0.})
        self.assertEqual(r,paired_summary(x,baseline=x,seed=3,replicates=512,alpha=.05))

    def test_extremes_and_draw_scoring(self):
        r=paired_summary(np.ones((12,2)),baseline=np.zeros((12,2)),seed=1,replicates=64,alpha=.01)
        self.assertEqual(r['improvement'],{'mean':1.,'lower':1.,'upper':1.})
        d=paired_summary(np.full((12,2),.5),seed=2,replicates=64,alpha=.05)
        self.assertEqual(d['draws'],24);self.assertEqual(d['mean'],.5)

    def test_reject_malformed_inputs(self):
        x=np.zeros((4,2))
        for bad in [[],[0,1],np.zeros((1,2)),np.zeros((3,3)),np.full((4,2),np.nan),np.full((4,2),.1),np.ones((4,2),dtype=bool)]:
            with self.subTest(bad=str(bad)),self.assertRaises(ValueError): paired_summary(bad,seed=1,replicates=64,alpha=.05)
        for kw in [{'seed':-1},{'seed':True},{'replicates':0},{'replicates':True},{'alpha':0},{'alpha':1},{'alpha':float('nan')},{'baseline':np.zeros((3,2))}]:
            args={'seed':1,'replicates':64,'alpha':.05};args.update(kw)
            with self.subTest(kw=kw),self.assertRaises(ValueError):paired_summary(x,**args)

        self.assertEqual(len(r['bins']),10);self.assertEqual(sum(x['count'] for x in r['bins']),5)
        self.assertEqual(r['bins'][0]['count'],2);self.assertEqual(r['bins'][9]['count'],2)
        self.assertEqual(r['bins'][1]['mean_prediction'],None)
        for v,y in [([],[]),([float('inf')],[0]),([0],[.1]),([0],[0,1])]:
            with self.assertRaises(ValueError):calibration(v,y)
