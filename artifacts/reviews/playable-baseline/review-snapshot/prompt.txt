22: // A ranking heuristic, NOT a damage calculator: no hidden items, abilities, stats or moves are inferred as facts.
23: function moveScore(moveId, own, foe, {tera = null, advertised = null} = {}) {
24:   const move = Dex.moves.get(moveId);
25:   if (!move.exists) return 1;
26:   const fraction = hpFraction(own.condition);
27:   if (move.category === 'Status') {
28:     if ((move.heal || move.id === 'rest') && fraction < 0.45 && !own.condition?.includes(' slp')) return 95 * (1 - fraction);
29:     return 4; // A deliberately weak fallback; no status/setup search in this baseline.
30:   }
31:   let type = advertised?.type ?? move.type;
32:   if (move.id === 'terablast' && (tera || own.tera || own.terastallized)) type = tera || own.tera || own.terastallized;
33:   const power = advertised?.basePower || move.basePower || (move.damage ? 70 : 50);
34:   const natural = Dex.species.get(speciesOf(own)).types;
35:   const current = typesOf(own);
36:   let stab = current.includes(type) ? 1.5 : 1;
37:   const teraType = tera || own.tera || own.terastallized;
38:   if (teraType && teraType !== 'Stellar') {
39:     stab = natural.includes(type) ? 1.5 : 1;
40:     if (type === teraType) stab = natural.includes(type) ? 2 : 1.5;
41:   }
42:   let offense = own.stats?.[move.category === 'Physical' ? 'atk' : 'spa'] ?? 100;
43:   if (move.category === 'Physical' && own.condition?.includes(' brn')) offense *= 0.5;
44:   const accuracy = move.accuracy === true ? 1 : move.accuracy / 100;
45:   return power * stab * effectiveness(type, foe) * accuracy * (offense / 100) + Math.max(0, move.priority) * 3;

54: export function chooseHeuristic(request, state) {
55:   const choices = legalChoices(request);
56:   if (!choices.length) return null;
57:   if (request.teamPreview || choices[0] === 'pass') return choices[0];
58:   const seat = request.side.id, foe = state.active[seat === 'p1' ? 'p2' : 'p1'];
59:   const team = request.side.pokemon;
60:   if (request.forceSwitch) {
61:     return choices.map(choice => [choice, switchScore(team[Number(choice.split(' ')[1]) - 1], foe)])
62:       .sort((a, b) => b[1] - a[1])[0][0];
63:   }
64:   const own = {...team.find(p => p.active), ...state.active[seat]};

66:   Object.assign(own, team.find(p => p.active));
