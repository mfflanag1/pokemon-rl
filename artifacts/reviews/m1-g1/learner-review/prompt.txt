19: def linked_gae(rewards, values, next_index, bootstrap, terminal, trajectory, *, gamma=1., lam=.95):
20:     """Next indices are same-player, same-episode decisions, ordered forward.
21: 
22:     next=-1 + terminal means no bootstrap; next=-1 + nonterminal is a buffer cut.
23:     At a cut bootstrap is that player's frozen-actor next value, never an opponent's.
24:     trajectory must distinguish both player AND episode; no alternating sign rule.
25:     """
26:     rewards, values, bootstrap = [np.asarray(a, dtype=F32) for a in (rewards, values, bootstrap)]
27:     next_index, terminal, trajectory = [np.asarray(a) for a in (next_index, terminal, trajectory)]
28:     n = len(rewards)
29:     require(n > 0 and all(a.shape == (n,) for a in (rewards, values, bootstrap, next_index, terminal, trajectory)), 'GAE shapes')
30:     require(next_index.dtype.kind in 'iu' and terminal.dtype == np.bool_, 'GAE index/terminal types')
31:     require(0 <= gamma <= 1 and 0 <= lam <= 1, 'GAE discounts')
32:     finite(rewards, values, bootstrap)
33:     advantages = np.zeros(n, dtype=F32)
34:     for i in range(n-1, -1, -1):
35:         j = int(next_index[i])
36:         require(j == -1 or i < j < n, 'GAE linkage must be forward or boundary')
37:         if terminal[i]:
38:             require(j == -1 and bootstrap[i] == 0, 'Terminal must not bootstrap')
39:             v_next, a_next = F32(0), F32(0)
40:         elif j == -1:
41:             v_next, a_next = bootstrap[i], F32(0)
42:         else:
43:             require(trajectory[i] == trajectory[j], 'Cross-player/episode linkage')
44:             require(bootstrap[i] == 0, 'Internal decision cannot have boundary bootstrap')
45:             v_next, a_next = values[j], advantages[j]
46:         delta = rewards[i] + F32(gamma) * v_next - values[i]
47:         advantages[i] = delta + F32(gamma * lam) * a_next
48:     returns = advantages + values
49:     finite(advantages, returns)
50:     return advantages, returns

53: def distribution(logits, mask):
54:     logits, mask = np.asarray(logits), np.asarray(mask)
55:     require(logits.dtype == F32 and logits.ndim == 2 and mask.shape == logits.shape and mask.dtype == np.bool_, 'Policy shape/dtype')
56:     require(len(logits) > 0 and np.all(mask.any(axis=1)), 'Empty legal-action mask')
57:     finite(logits)
58:     maxima = np.max(np.where(mask, logits, F32(-np.inf)), axis=1, keepdims=True)
59:     # Illegal entries have exactly zero probability and a finite log sentinel.
60:     shifted = np.where(mask, logits - maxima, F32(0))
61:     exp = np.where(mask, np.exp(shifted), F32(0))
62:     denominator = exp.sum(axis=1, keepdims=True, dtype=F32)
63:     logp = np.where(mask, shifted - np.log(denominator), F32(0))
64:     probabilities = exp / denominator

69: def ppo_loss_grad(logits, values, mask, actions, old_logp, advantages, returns, *, clip=.2, value_coef=.5, entropy_coef=0.):
70:     """Mean clipped actor loss + coefficient * half-MSE - coefficient * entropy."""
71:     p, logp = distribution(logits, mask)
72:     n, actions_n = p.shape
73:     values, old_logp, advantages, returns = [np.asarray(a) for a in (values, old_logp, advantages, returns)]
74:     actions = np.asarray(actions)
75:     require(all(a.shape == (n,) and a.dtype == F32 for a in (values, old_logp, advantages, returns)), 'PPO FP32 vectors')
76:     require(actions.shape == (n,) and actions.dtype.kind in 'iu' and np.all((actions >= 0) & (actions < actions_n)), 'PPO action indices')
77:     require(np.all(mask[np.arange(n), actions]), 'Known-forbidden action')
78:     require(0 < clip < 1 and value_coef >= 0 and entropy_coef >= 0, 'PPO coefficients')
79:     finite(values, old_logp, advantages, returns)
80:     log_ratio = logp[np.arange(n), actions] - old_logp
81:     with np.errstate(over='raise', invalid='raise'):
82:         ratio = np.exp(log_ratio)
83:     clipped = np.clip(ratio, F32(1-clip), F32(1+clip))
84:     actor_loss = -np.minimum(ratio * advantages, clipped * advantages).mean(dtype=F32)
85:     entropy_rows = -(p * logp).sum(axis=1, dtype=F32)

87:     value_loss = F32(.5) * np.square(values - returns).mean(dtype=F32)
88:     loss = actor_loss + F32(value_coef) * value_loss - F32(entropy_coef) * entropy
89:     active = np.where(advantages >= 0, ratio <= F32(1+clip), ratio >= F32(1-clip))
90:     score = -p.copy(); score[np.arange(n), actions] += F32(1)
91:     grad_logits = (-advantages * ratio * active / F32(n))[:, None] * score

93:     grad_values = F32(value_coef / n) * (values - returns)
94:     finite(loss, ratio, grad_logits, grad_values)
95:     return {'loss': float(loss), 'actor_loss': float(actor_loss), 'value_loss': float(value_loss),
96:             'entropy': float(entropy), 'max_ratio_error': float(np.max(np.abs(ratio-F32(1)))),
97:             'clip_fraction': float(np.mean(np.abs(ratio-F32(1)) > clip)),

112:         finite(*parameters.values(), *gradients.values(), *self.m.values(), *self.v.values())
113:         norm = float(np.sqrt(sum(np.square(g.astype(np.float64)).sum() for g in gradients.values())))
114:         scale = F32(min(1., self.max_norm / max(norm, 1e-30)))
115:         self.step += 1
116:         for k, gradient in gradients.items():

118:             self.m[k] = F32(.9) * self.m[k] + F32(.1) * g

123:         finite(*parameters.values(), *self.m.values(), *self.v.values())

1: # Proposed execution gates and experiment ladder
2: 
3: Contribution for the execution plan, 2026-10-10. All numbers below are proposed engineering acceptance criteria, not facts recovered from Jaxcalibur or guarantees of completeness/convergence. A gate combines an explicit supported scope, executable checks, and logged evidence. Passing a statistical screen licenses the next experiment; it does not prove global correctness or optimality.
4: 
5: ## Dependency order and safe parallelism
6: 
7: The critical path is: pinned format/data contract → request/observation semantics → reference parity harness → restricted JAX engine → correct small PPO learner → measured learning → expanded mechanics → complete target-format evaluation → search. Keep action IDs, observation schemas, phase names, and checkpoint versions stable early.
8: 
9: Three independent workstreams are safe after the interfaces are agreed:
10: 
11: - **Simulator:** mechanics, reference scenarios, request timing, supported-feature manifest, JAX throughput. Owns `FullState` and transitions.
12: - **Policy/learner:** small transformer, per-player rollout buffer, PPO/GAE, auxiliary heads, checkpoint/resume. Initially develops against tiny scripted environments and frozen observation fixtures. Owns learner internals, not simulator truth extraction.
13: - **Data/evaluation:** pinned generator export, sampled-team validation, frozen team seeds, baseline policies, scoring and paired intervals. Owns fixtures and evaluation reports, not engine mechanics.
14: 
15: A separate observation/adapter owner approves interface changes spanning these streams. Belief-model design can run in parallel once observation fixtures exist; connecting beliefs to production action features waits for leakage tests. Search literature/toy-game experiments can run in parallel, but integration waits for a useful validated policy and simulator. Do not launch a large training run while known mechanics divergences remain in its sampled scope.
16: 
17: ## Coverage stages: call each game what it is
18: 
19: **Stage A: synthetic canaries.** One-step bandits, deterministic delayed reward, asymmetric forced replacements, and toy hidden-information games. These test algorithms, not Pokémon strength.
20: 
21: **Stage B: restricted Pokémon.** Explicit fixed teams/sets and a small mechanics whitelist, first 1v1 then 3v3/6v6. Use reference battles with those exact teams and documented rules. Do not call a whitelist-filtered training distribution unmodified Gen 9 Random Battles. Excluded features must be impossible to encounter or must fail loudly before sample collection; never implement unsupported effects as silent no-ops.
22: 
23: **Stage C: expanding mechanics.** Add complete mechanic families in batches, with interactions: status/residual damage; stat changes; switching/pivots/trapping; item/ability interactions; field effects; Tera; copying/transformation and other edge cases. Re-run prior fixtures and a newly sampled interaction suite after each batch. Keep the reduced-format baseline checkpoint and compare transfer against training from scratch at the same budget.
24: 
25: **Stage D: intended complete format.** Pin the exact official generator, rules, simulator revision, and data export. Audit all generated sets *and the transitive mechanics they can invoke*; move copying or random move effects can extend beyond listed sets. A closed generator manifest plus a passing harness is evidence of supported scope, not proof that every interaction is correct. Record unknowns and observed unsupported effects as blockers to an unqualified full-format claim. Default train/eval sampling may only include fully supported matches; if exclusions remain, label the result restricted and report rejection rates/distribution shift.
26: 
27: ## Quantitative gate ledger
28: 
29: ### G0 — Contracts and testability (before learning)
30: 
31: Deliver a reproducible version manifest; phase diagram; observable action-mask contract; full-state/observation separation; deterministic seed protocol; supported-feature manifest; saved reference fixtures; checkpoint schema.
32: 
33: Acceptance: every phase has a fixture for normal entry, terminal exit, cancellation/re-request if applicable, and both player roles. Every action ID round-trips between network, simulator, and adapter. All displayed/consumed observations must be reconstructible from the same player's protocol history and private information. Unknown enum values and unsupported mechanics fail visibly. Zero tolerated silent fallbacks that alter battle rules.
34: 
35: ### G1 — Learner canaries (before Pokémon scale)
36: 
37: Run on CPU/FP32 where practical, using a transparent reference implementation.
38: 
39: - GAE/returns/log-ratios agree with hand-computed short trajectories within absolute 1e-5; terminal and buffer-cut cases are separate. No terminal bootstrap; rollout truncation bootstraps from the correct perspective.
40: - Same weights/data before an update give PPO ratios of 1 within 1e-5. Single-legal-action decisions have actor gradient/entropy zero within 1e-7, while a forced switch with multiple candidates remains trainable. The scalar surrogate need not be zero (it can equal minus the advantage); alternatively mask those rows out of the actor objective explicitly.
41: - One-step four-action canary, with exploration penalties disabled: all three learner seeds put at least 99% probability on the known rewarding action within 100,000 sampled decisions. This is a diagnostic target; failure triggers learner investigation rather than immediately expanding training.
42: - Deterministic delayed-reward canary learns both player perspectives without winner/loser sign inversion. Every example with a known action forbidden by the observable mask produces zero probability for that action. No NaN/Inf in sampled logits, probabilities, losses, gradients, or optimizer state.
43: - The boundary-checkpoint resume tests below pass before jobs longer than two hours. Mid-collection/mid-epoch tests become required only if those continuation modes are implemented.
44: 
45: ### G2 — Restricted engine fidelity and information safety
46: 
47: Acceptance within the declared scope:
48: 
49: - Zero unexplained disagreements in legal request structure, observation/reveal timing, HP/status/PP changes, winner, and phase transitions across all hand-authored fixtures and at least 10,000 independently seeded reference battles. Adapt the battle count to CPU cost, but record the actual count and do not substitute correlated transitions for independent games in a failure-rate claim.
50: - Deterministic submechanics and forced random outcomes should compare exactly after normalization. For chance mechanics, either enumerate finite outcome distributions or run predeclared probability checks with a family-wise error policy. Same seed alone is not a parity contract if the engines consume randomness differently; inject/check semantically identified chance outcomes.
51: - At least 1,000 paired hidden-state mutations, stratified over unrevealed species/sets/items/abilities/Tera and request phases: normalized player observations must be identical whenever that player's information is identical. Deterministic inference logits must then match within 1e-6 FP32 (or exactly for byte-identical inputs and deterministic backend). This includes damage features, memory, masks, and cached values.
52: - For each covered mechanics family, include adversarial boundary cases: HP/KO thresholds, move PP exhaustion, simultaneous fainting, interrupted turns, forced switches, stale requests, and legal-information changes. Random volume cannot replace these targeted tests.
53: 
54: Interpretation: zero failures in 10,000 independent battles implies an approximate 95% upper bound of 0.03% for the tested distribution's per-battle failure probability, *not* for an untested distribution or all mechanics. A discovered divergence blocks training on affected states until resolved or explicitly excluded.
55: 
56: ### G3 — Stable restricted-policy pilot
57: 
58: Use the smallest practical feed-forward/history-token transformer first; defer recurrence and specialized token MLPs. Measure complete collect/update cycles, not isolated kernel throughput. Proposed pilot: three training seeds, a modest predeclared decision budget, and checkpoint evaluation every fixed budget increment.
59: 
60: Hard gates: zero unsupported effects or observation-mask bugs in training scope; zero NaN/Inf; no unexpected recompilation after warm-up at fixed shapes; a two-hour soak without OOM; peak allocated memory leaves roughly 15% headroom where the selected runtime permits it. Throughput gate is budget-derived: measured decisions/s must project the next experiment within its allocated GPU hours with at least 25% overhead allowance. Report decisions/game as well as games/s.
61: 
62: Learning gate: reliable improvement over initialization against a frozen baseline panel, plus a predeclared starter target such as point score ≥60% against random legal play and lower 95% confidence bound >55%. Include a damage-greedy baseline and old checkpoints as diagnostics. Beating random alone does not authorize a claim of strategic competence. Analyze all three training seeds; a single successful seed is insufficient to conclude the learner is stable. Record KL/clip fraction, entropy, legal-action probability floor, critic calibration, phase-specific losses, battle length, and timeouts.
63: 
64: ### G4 — Full intended-format readiness
65: 
66: Expand G2's parity/hidden-info checks to all supported format families and the pinned official team distribution. Run at least 10,000 fresh independent reference battles after the last mechanics/schema change, and an adapter soak of at least 1,000 private/local evaluation games. Require zero unexplained divergences, zero known-forbidden actions, zero unsupported mechanics, and zero server-visible phase/request corruption. Distinguish legitimate hidden-legality rejections from client mistakes.
67: 
68: A production-length run is authorized technically only after a short full-format learner smoke test and resume test pass, and its measured complete-cycle throughput/memory predict the planned budget. Save learning curves against fixed opponents; current self-play score near 50% is not a learning metric.
69: 
70: ### G5 — Search integration (after a useful policy)
71: 
72: First baseline: world-averaged one-step lookahead versus an observation-only opponent policy. Then simultaneous deeper search. Before Pokémon integration, test 2×2 games and rock-paper-scissors against exact solutions: at the declared rollout budget, mean-strategy exploitability ≤0.02 on normalized [-1,1] payoffs across a fixed test suite and at least 20 seeds is a proposed target. Report the whole distribution and worst case; do not elevate toy-game success into a Pokémon equilibrium claim.
73: 
74: Require exact pre-reveal hidden-information invariance for any opponent policy component explicitly intended to use only that opponent's information. Test returned action distributions, not only network inputs, because optimized search backups can leak secrets. Root mixed-strategy sampling is the default candidate; argmax is an ablation.
75: 
76: Promote a search configuration only after a predeclared meaningful gain (for example ≥2 percentage points in direct head-to-head score) with a paired lower confidence bound above 50%, no material regression against the fixed opponent panel, and p99 decision latency under the adapter's specified time budget. Use measured latency bounds chosen from the target deployment; do not invent a universal seconds threshold. Search timeouts must return a legal base-policy action and be counted.
77: 
78: ## Phase-aware reject/re-request semantics
79: 
80: A policy's action mask represents what the player is allowed to infer from its request, not all legality knowable to the omniscient simulator. A choice may be admissible under that request but rejected when hidden mechanics are checked. Masking it using hidden truth can disclose information.
81: 
82: Implement and test three classes:
83: 
84: 1. **Known-invalid choice:** forbidden by the observable mask, stale request ID, malformed mapping, or wrong phase. This is an implementation bug; quarantine/log the affected sample. Do not learn from accidental protocol failures.
85: 2. **Rule-defined hidden-legality rejection:** reproduce the reference rejection message and knowledge update, issue the appropriate new request, retain commitments exactly as the protocol requires, and consume no turn/PP/chance/reward unless the reference does. The attempted choice and resulting information-bearing re-request form a zero-reward transition in that player's decision trajectory; do not silently erase useful feedback or reset GAE as if the game ended.
86: 3. **Transport retry:** replay idempotently against the request ID; do not create a new learner decision or change the opponent's commitment merely because a message was resent.
87: 
88: Fixtures must cover repeated rejected choices, simultaneous submissions, one player awaiting re-request while the other has committed, terminal events before an anticipated request, and a hidden restriction becoming known. Test whether each feedback event is public or private. Keep the rejection/phase state in checkpoints. Retry-count safeguards should surface a bug or explicit failure, not secretly adjudicate a match as lost.
89: 
90: ## Resume and on-policy state
91: 
92: Start with synchronous collection: a frozen actor snapshot collects a buffer; PPO updates that buffer; the next collection uses the new actor. Multiple GPUs, async actors, and stale-policy correction can wait.
93: 
94: A continuation checkpoint includes learner parameters, optimizer/scaler state, schedules and update counters; behavior/actor parameters and version; live simulator states; phase/request IDs and pending actions; observations/private memory; all PRNG streams; partially filled buffers with old log-probabilities/values/masks; collection cursor; minibatch permutation and epoch/minibatch cursor (plus accumulated gradients if mid-accumulation saving is supported); and data/schema/source versions. Save atomically with checksums. A manifest must make partial writes distinguishable from valid checkpoints.
95: 
96: Initial resume gate: support completed-update checkpoints at the documented collection barrier, and on a deterministic CPU canary reproduce the next samples and updates from a restore.
