2: 
3: The rare FP32 mass and its real complement define the law. Logs use log/log1p

18: def parts(logits, mask):
19:     both = jnp.all(mask, axis=1)
20:     # Forced/padded rows must have no logit gradient, including their difference.
21:     safe = jnp.where(both[:, None], logits, jnp.float32(0))
22:     delta = safe[:, 0] - safe[:, 1]
23:     rare_action = jnp.where(delta >= 0, 1, 0).astype(jnp.int32)
24:     tail = jnp.exp(-jnp.abs(delta))
25:     raw = tail / (jnp.float32(1) + tail)
26:     rare = jnp.where(raw > FLOOR, raw, FLOOR)  # zero derivative at the floor
27:     is_rare = jnp.arange(2)[None, :] == rare_action[:, None]
28:     p = jnp.where(is_rare, rare[:, None], jnp.float32(1) - rare[:, None])
29:     logs = jnp.where(is_rare, jnp.log(rare)[:, None], jnp.log1p(-rare)[:, None])
30:     # Forced legal action: exact one/zero; terminal: exact all-zero transport.
31:     return (jnp.where(both[:, None], p, mask.astype(jnp.float32)),
32:             jnp.where(both[:, None], logs, jnp.float32(0)), rare_action, rare)

6: def masked_loss(logits, values, mask, actions, old_logp, advantages, returns, valid,
7:                 clip=.2, value_coef=.5, entropy_coef=.01):
8:     """Padded PPO loss -> (scalar_loss, metrics) over the rows flagged by `valid`.
9: 
10:     Shapes (n rows, A actions): `logits` (n, A) FP32, `values` (n,) FP32, `mask` (n, A) bool,
11:     `actions` (n,) integer, `old_logp`/`advantages`/`returns` (n,) FP32, `valid` (n,) bool.
12:     `clip`, `value_coef`, `entropy_coef` are Python floats (host-rounded to FP32 like the
13:     reference's `F32(1 - clip)` / `F32(1 + clip)`); treat them as static under `jax.jit`.
14: 
15:     Reduction and clipping semantics on valid rows follow the sealed
16:     `pokemon_jax_learner.core._loss`; distribution is binary-policy-1:
17:     mean clipped actor loss with the explicit inclusive `active` branch (equality on the
18:     unclipped side keeps the gradient), `value_coef` * half-MSE value loss, minus
19:     `entropy_coef` * mean entropy; `old_logp`, `advantages`, `returns` are stop-gradient.
20:     Metrics are the reference's seven keys: `loss`, `actor_loss`, `value_loss`, `entropy`,
21:     `max_ratio_error` (max over valid rows), `clip_fraction`, `approx_kl` (valid-row means).
22: 
23:     Invalid rows may hold any transport payload (NaN logits/values/targets, action -1,
24:     all-false mask). Every input is sanitized with `jnp.where(valid, ...)` to neutral finite
25:     values (zeros; all-true mask; action 0) *before* any arithmetic, and every per-row term is
26:     re-masked before its reduction, so invalid rows contribute neither value nor gradient:
27:     both selections have exact-zero cotangents, giving exactly 0 gradient entries on padded
28:     rows. With no valid rows the divisor is 1, so all seven metrics and both gradients are
29:     exactly zero. Differentiable in `logits` and `values`; no host checks.
30:     """
31:     valid = jnp.asarray(valid, bool)
32:     valid_rows = valid[:, None]
33:     zero = jnp.float32(0)
34:     # Sanitize *all* inputs on invalid rows before any arithmetic touches them.
35:     logits = jnp.where(valid_rows, logits, zero)
36:     mask = jnp.where(valid_rows, mask, True)
37:     actions = jnp.where(valid, actions, 0).astype(jnp.int32)
38:     values, old_logp, advantages, returns = [jnp.where(valid, a, zero) for a in (values, old_logp, advantages, returns)]
39:     old_logp, advantages, returns = [jax.lax.stop_gradient(a) for a in (old_logp, advantages, returns)]
40:     low, high, clip, value_coef, entropy_coef = [
41:         jnp.asarray(c, jnp.float32) for c in (1 - clip, 1 + clip, clip, value_coef, entropy_coef)]
42: 
43:     count = jnp.maximum(jnp.sum(valid.astype(jnp.float32)), jnp.float32(1))
44: 
45:     def valid_mean(rows):
46:         return jnp.sum(jnp.where(valid, rows, zero)) / count
47: 
48:     p, logp = distribution_kernel(logits, mask)
49:     n = logits.shape[0]
50:     log_ratio = logp[jnp.arange(n), actions] - old_logp
51:     ratio = jnp.exp(log_ratio)
52:     clipped = jnp.clip(ratio, low, high)
53:     # Reference inclusive branch: equality on the unclipped side keeps the gradient.
54:     active = jnp.where(advantages >= 0, ratio <= high, ratio >= low)
55:     surrogate = jnp.where(active, ratio, jax.lax.stop_gradient(clipped)) * advantages
56:     actor_loss = -valid_mean(surrogate)
57:     entropy = valid_mean(-jnp.sum(p * logp, axis=1))
58:     value_loss = jnp.float32(.5) * valid_mean(jnp.square(values - returns))
59:     loss = actor_loss + value_coef * value_loss - entropy_coef * entropy
60:     error = jnp.abs(ratio - jnp.float32(1))
61:     metrics = {'loss': loss, 'actor_loss': actor_loss, 'value_loss': value_loss, 'entropy': entropy,
62:                'max_ratio_error': jnp.max(jnp.where(valid, error, zero)),
63:                'clip_fraction': valid_mean((error > clip).astype(jnp.float32)),
64:                'approx_kl': valid_mean((ratio - jnp.float32(1)) - log_ratio)}
65:     return loss, metrics

26: def _checked_probability(probability: object) -> np.float32:
27:     if not isinstance(probability, np.float32):
28:         raise TypeError("probability must be an actual numpy.float32 scalar")
29:     if not np.isfinite(probability):
30:         raise ValueError("probability must be finite")
31:     if probability < MINIMUM_LEGAL_MASS or probability > np.float32(0.5):
32:         raise ValueError("probability must be in [2^-24, 0.5]")
33:     return probability
34: 
35: 
36: def dyadic_threshold(probability: object) -> tuple[int, int]:
37:     """Return high/low uint32 words of the exact integer ``p * 2**64``."""
38: 
39:     probability = _checked_probability(probability)
40:     numerator, denominator = probability.item().as_integer_ratio()
41:     scaled = Fraction(numerator, denominator) * (1 << 64)
42:     if scaled.denominator != 1:  # Defensive: impossible for an in-range float32.
43:         raise ValueError("probability does not have an integral 64-bit threshold")
44:     threshold = scaled.numerator
45:     return threshold >> 32, threshold & _UINT32_MAX
46:

48: def _checked_word(word: object, name: str) -> int:
49:     # bool is intentionally excluded even though it subclasses int.
50:     if type(word) is not int:
51:         raise TypeError(f"{name} must be a Python int")
52:     if not 0 <= word <= _UINT32_MAX:
53:         raise ValueError(f"{name} must be in uint32 range")
54:     return word

57: def word_selects_rare(high: object, low: object, probability: object) -> bool:
58:     """Whether the concatenated high/low word lies below the exact threshold."""
59: 
60:     high = _checked_word(high, "high")
61:     low = _checked_word(low, "low")
62:     threshold_high, threshold_low = dyadic_threshold(probability)
63:     word = (high << 32) | low
64:     threshold = (threshold_high << 32) | threshold_low
65:     return word < threshold

84: def _rare_mass(logits: np.ndarray) -> tuple[int, np.float32, np.float32]:
85:     d = np.float32(logits[0] - logits[1])
86:     rare_index = 1 if d >= np.float32(0.0) else 0
87:     exponent = np.float32(-np.abs(d))
88:     exp_value = np.exp(exponent, dtype=np.float32)
89:     raw = np.float32(exp_value / np.float32(1.0 + exp_value))
90:     return rare_index, np.maximum(raw, MINIMUM_LEGAL_MASS), raw

93: def scalar_policy(logits: object, mask: object) -> ScalarPolicy:
94:     """Evaluate the contract's clipped two-action policy for one state."""
95: 
96:     logits, mask = _checked_policy_inputs(logits, mask)
97:     legal = int(mask[0]) + int(mask[1])
98:     probabilities = np.zeros(2, dtype=np.float32)
99:     logp = np.zeros(2, dtype=np.float32)
100:     if legal == 0:
101:         return ScalarPolicy(probabilities, logp, -1, np.float32(0.0))
102:     if legal == 1:
103:         only = 0 if mask[0] else 1
104:         probabilities[only] = np.float32(1.0)
105:         return ScalarPolicy(probabilities, logp, only, np.float32(0.0))
106: 
107:     rare_index, rare_mass, _ = _rare_mass(logits)
108:     common_index = 1 - rare_index
109:     probabilities[rare_index] = rare_mass
110:     probabilities[common_index] = np.float32(np.float32(1.0) - rare_mass)
111:     logp[rare_index] = np.log(rare_mass, dtype=np.float32)
112:     logp[common_index] = np.log1p(np.float32(-rare_mass), dtype=np.float32)
113:     return ScalarPolicy(probabilities, logp, rare_index, rare_mass)

123:         raise TypeError("selected_action must be a Python int")
